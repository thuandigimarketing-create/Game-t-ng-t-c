            <div class="coin-reward">🪙 +10</div>
            <button class="btn" id="next-level-btn" onclick="nextLevel()">Sang Cấp Tiếp Theo</button>

    <script>
        // Các thiết lập của trò chơi
        let playerName = "";
        let currentLevel = 1;
        let coins = 0;
        let timeLeft = 60;
        let timerInterval;
        let currentWordIndex = 0;

        // Bộ từ vựng theo cấp độ cho học sinh lớp 3
        const levels = {
            1: ["a", "s", "d", "f", "j", "k", "l", "h", "g", "q", "w", "e", "r"], // Làm quen phím
            2: ["ca", "meo", "cho", "hoc", "tap", "ban", "ghe", "sach", "vo", "truong"], // Từ không dấu cơ bản
            3: ["cá voi", "con mèo", "học sinh", "lớp ba", "trường học", "bạn bè", "chăm ngoan"] // Cụm từ
        };
        let currentWords = [];

        // Hệ thống âm thanh (Sử dụng Tone.js)
        const synth = new Tone.Synth().toDestination(); // Âm thanh gõ phím
        const polySynth = new Tone.PolySynth().toDestination(); // Âm thanh chúc mừng

        function playKeySound() {
            synth.triggerAttackRelease("C5", "32n"); // Tiếng "tíc" nhẹ
        }

        function playCorrectSound() {
            synth.triggerAttackRelease("E5", "8n"); // Tiếng "ting" cao hơn
        }

        function playWinSound() {
            const now = Tone.now();
            polySynth.triggerAttackRelease(["C4", "E4", "G4", "C5"], "2n", now); // Hợp âm chiến thắng
        }

        // Bắt đầu trò chơi
        async function startGame() {
            // Yêu cầu khởi động Audio Context cho trình duyệt
            await Tone.start();
            
            const nameInput = document.getElementById("player-name").value;
            if (!nameInput) {
                alert("Bé nhớ nhập tên nhé!");
                return;
            }
            
            playerName = nameInput;
            document.getElementById("display-name").innerText = "👤 " + playerName;
            document.getElementById("login-screen").classList.add("hidden");
            document.getElementById("game-screen").classList.remove("hidden");
            
            loadLevel(1);
        }

        // Tải dữ liệu cấp độ
        function loadLevel(level) {
            currentLevel = level;
            document.getElementById("level-display").innerText = currentLevel;
            currentWords = [...levels[currentLevel]];
            // Trộn ngẫu nhiên từ
            currentWords.sort(() => Math.random() - 0.5); 
            currentWordIndex = 0;
            
            timeLeft = 60;
            document.getElementById("time-left").innerText = timeLeft;
            
            showNextWord();
            document.getElementById("type-input").value = "";
            document.getElementById("type-input").focus();
            
            clearInterval(timerInterval);
            timerInterval = setInterval(updateTimer, 1000);
        }

        // Hiển thị từ tiếp theo
        function showNextWord() {
            if (currentWordIndex >= currentWords.length) {
                levelComplete();
                return;
            }
            document.getElementById("target-word").innerText = currentWords[currentWordIndex];
        }

        // Cập nhật đồng hồ đếm ngược
        function updateTimer() {
            timeLeft--;
            document.getElementById("time-left").innerText = timeLeft;
            if (timeLeft <= 0) {
                clearInterval(timerInterval);
                alert("Hết giờ rồi! Hãy thử lại nhé.");
                loadLevel(currentLevel); // Chơi lại cấp độ hiện tại
            }
        }

        // Xử lý sự kiện gõ phím
        const inputField = document.getElementById("type-input");
        inputField.addEventListener("input", function(e) {
            playKeySound(); // Phát âm thanh mỗi khi gõ
            
            const currentWord = currentWords[currentWordIndex];
            const typedWord = inputField.value;

            if (typedWord === currentWord) {
                playCorrectSound();
                coins += 2; // Cộng 2 xu cho mỗi từ đúng
                document.getElementById("coin-count").innerText = coins;
                currentWordIndex++;
                inputField.value = "";
                showNextWord();
            }
        });

        // Hoàn thành cấp độ
        function levelComplete() {
            clearInterval(timerInterval);
            playWinSound(); // Phát nhạc chúc mừng
            coins += 10; // Thưởng 10 xu qua màn
            document.getElementById("coin-count").innerText = coins;
            
            document.getElementById("game-screen").classList.add("hidden");
            document.getElementById("celebration-screen").classList.remove("hidden");
            document.getElementById("winner-name").innerText = playerName;
            
            if (currentLevel >= 3) {
                document.getElementById("celebration-message").innerText = "Bé đã phá đảo toàn bộ trò chơi! Xuất sắc!";
                document.getElementById("next-level-btn").innerText = "Chơi lại từ đầu";
            } else {
                document.getElementById("celebration-message").innerText = `Bé đã vượt qua Cấp ${currentLevel}! Tặng bé 10 đồng xu.`;
                document.getElementById("next-level-btn").innerText = "Sang Cấp Tiếp Theo";
            }
        }

        // Chuyển cấp
        function nextLevel() {
            document.getElementById("celebration-screen").classList.add("hidden");
            document.getElementById("game-screen").classList.remove("hidden");
            
            if (currentLevel >= 3) {
                coins = 0;
                document.getElementById("coin-count").innerText = coins;
                loadLevel(1);
            } else {
                loadLevel(currentLevel + 1);
            }
        }
    </script>
